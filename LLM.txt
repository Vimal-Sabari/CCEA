def parse_csv(csv_path: Path) -> List[CsvRow]:
    """
    Parse data.csv using header names to find columns (not fixed positions).
    Works regardless of delimiter style (single comma, double comma, etc.).
    """
    rows: List[CsvRow] = []
    with open(csv_path, "r", encoding="utf-8-sig") as f:
        # Bug 7: delimiter is hardcoded to ',' which is correct for the current
        # data.csv. If the file ever uses a different delimiter, update here.
        reader = csv.reader(f, delimiter=",")
        header = next(reader, None)
        if not header:
            print(f"[CSV] WARN: {csv_path.name} is empty or contains no header.")
            return []

        # Build column map by header name (skip empty column names)
        col_map: Dict[str, int] = {}
        for i, h in enumerate(header):
            name = h.strip()
            if name:
                col_map[name] = i

        print(f"[CSV] Header detected ({len(header)} raw columns): {header}")
        print(f"[CSV] Named columns: {col_map}")

        required = ["Signature_Name", "Defenses", "Initial_Severity"]
        for req in required:
            if req not in col_map:
                print(f"[X] ERROR: Required column '{req}' not found in header.")
                print(f"    Available: {list(col_map.keys())}")
                sys.exit(1)

        sig_idx = col_map["Signature_Name"]
        def_idx = col_map["Defenses"]
        sev_idx = col_map["Initial_Severity"]
        sig_desc_idx = col_map.get("Signature_Description")
        tactics_idx = col_map.get("Tactics")
        tech_idx = col_map.get("Technique")
        def_desc_idx = col_map.get("Defense_Description")

        for line_num, raw_row in enumerate(reader, start=2):
            if not raw_row or all(cell.strip() == "" for cell in raw_row):
                continue

            try:
                sig_name = raw_row[sig_idx].strip()
                defense = raw_row[def_idx].strip()
                severity_str = raw_row[sev_idx].strip()

                if not sig_name or not defense or not severity_str:
                    print(f"[CSV] WARN: Skipping incomplete row at line {line_num}")
                    continue

                initial_severity = int(severity_str)
                sig_desc = raw_row[sig_desc_idx].strip() if sig_desc_idx is not None and sig_desc_idx < len(raw_row) else ""
                tactics = raw_row[tactics_idx].strip() if tactics_idx is not None and tactics_idx < len(raw_row) else ""
                technique = raw_row[tech_idx].strip() if tech_idx is not None and tech_idx < len(raw_row) else ""
                def_desc = raw_row[def_desc_idx].strip() if def_desc_idx is not None and def_desc_idx < len(raw_row) else ""

                rows.append(CsvRow(sig_name, defense, initial_severity, sig_desc, tactics, technique, def_desc))
            except (IndexError, ValueError) as e:
                print(f"[CSV] WARN: Error parsing line {line_num}: {e}")
                continue

    print(f"[CSV] Parsed {len(rows)} valid rows from {csv_path.name}")
    return rows

def _extract_string_value(node: ast.AST) -> Optional[str]:
    """Extract a string value from an AST Constant node."""
    if isinstance(node, ast.Constant) and isinstance(node.value, str):
        return node.value
    return None


def _extract_int_value(node: ast.AST) -> Optional[int]:
    """Extract an integer value from an AST Constant node."""
    if isinstance(node, ast.Constant) and isinstance(node.value, int):
        return node.value
    return None


def _extract_list_of_strings(node: ast.AST) -> List[str]:
    """Extract a list of strings from List/Tuple/Set AST nodes.
    Bug 5 fix: also handles bare string constants (e.g. filter_apinames = "NtCreateFile").
    """
    # Bare string constant: filter_apinames = "NtCreateFile"
    if isinstance(node, ast.Constant) and isinstance(node.value, str):
        return [node.value]
    if isinstance(node, (ast.List, ast.Tuple, ast.Set)):
        result = []
        for elt in node.elts:
            val = _extract_string_value(elt)
            if val is not None:
                result.append(val)
        return result
    # Handle set([...]) call
    if isinstance(node, ast.Call):
        if isinstance(node.func, ast.Name) and node.func.id == "set":
            if node.args:
                return _extract_list_of_strings(node.args[0])
    return []


def _extract_class_source(file_lines: List[str], class_node: ast.ClassDef) -> str:
    """Extract the exact source code of a class from file lines."""
    start = class_node.lineno - 1
    end = class_node.end_lineno if class_node.end_lineno else len(file_lines)
    return "".join(file_lines[start:end])


def _extract_attr_source(file_lines: List[str], node: ast.Assign) -> str:
    """Extract the raw source text of an assignment statement."""
    start = node.lineno - 1
    end = node.end_lineno if node.end_lineno else node.lineno
    return "".join(file_lines[start:end]).strip()


def build_signature_index(windows_dir: Path) -> Dict[str, SignatureInfo]:
    """
    Scan all .py files recursively in windows/ and build an index.
    Uses rglob for recursive discovery (Fix #9).
    Raises error on duplicate signature names (Fix #10).
    Extracts trigger-related attributes for structured prompt (Fix #14).
    """
    index: Dict[str, SignatureInfo] = {}
    py_files = sorted(windows_dir.rglob("*.py"))  # Fix #9: recursive
    skipped = 0

    for py_file in py_files:
        try:
            source = py_file.read_text(encoding="utf-8", errors="replace")
            file_lines = source.splitlines(keepends=True)
            tree = ast.parse(source, filename=str(py_file))
        except SyntaxError as e:
            print(f"[INDEX] WARN: Syntax error in {py_file.name}: {e}")
            skipped += 1
            continue
        except OSError as e:
            # Bug 8 fix: catch locked/unreadable files instead of crashing
            print(f"[INDEX] WARN: Cannot read {py_file.name}: {e}")
            skipped += 1
            continue

        # Collect module-level lists / dicts / indicators
        module_level_attrs: Dict[str, str] = {}
        for mod_item in tree.body:
            if isinstance(mod_item, ast.Assign):
                for target in mod_item.targets:
                    if isinstance(target, ast.Name):
                        if isinstance(mod_item.value, (ast.List, ast.Tuple, ast.Set, ast.Dict)):
                            module_level_attrs[target.id] = _extract_attr_source(file_lines, mod_item)

        for node in ast.walk(tree):
            if not isinstance(node, ast.ClassDef):
                continue

            sig_name = None
            description = ""
            severity = 1  # Cuckoo base class Signature default is 1
            categories: List[str] = []
            ttp: List[str] = []
            filter_apinames: List[str] = []
            filter_categories: List[str] = []
            trigger_attributes: Dict[str, str] = {}

            for item in node.body:
                if not isinstance(item, ast.Assign):
                    continue
                for target in item.targets:
                    if not isinstance(target, ast.Name):
                        continue
                    attr_name = target.id

                    if attr_name == "name":
                        sig_name = _extract_string_value(item.value)
                    elif attr_name == "description":
                        val = _extract_string_value(item.value)
                        if val is not None:
                            description = val
                    elif attr_name == "severity":
                        val = _extract_int_value(item.value)
                        if val is not None:
                            severity = val
                        # Bug 9: non-literal (e.g. severity = SEVERITY_HIGH)
                        # stays as sentinel -1 to distinguish from a genuine 0
                        elif not isinstance(item.value, ast.Constant):
                            severity = -1  # signals "non-literal, could not extract"
                    elif attr_name == "categories":
                        categories = _extract_list_of_strings(item.value)
                    elif attr_name == "ttp":
                        ttp = _extract_list_of_strings(item.value)
                    elif attr_name == "filter_apinames":
                        filter_apinames = _extract_list_of_strings(item.value)
                        trigger_attributes[attr_name] = _extract_attr_source(
                            file_lines, item
                        )
                    elif attr_name == "filter_categories":
                        filter_categories = _extract_list_of_strings(item.value)
                        trigger_attributes[attr_name] = _extract_attr_source(
                            file_lines, item
                        )
                    elif attr_name in TRIGGER_ATTR_NAMES:
                        # Store raw source for trigger-related attributes
                        trigger_attributes[attr_name] = _extract_attr_source(
                            file_lines, item
                        )

            # Check for module-level variables referenced in this class
            class_identifiers = set()
            for inner in ast.walk(node):
                if isinstance(inner, ast.Name):
                    class_identifiers.add(inner.id)
            for mod_var_name, mod_var_src in module_level_attrs.items():
                if mod_var_name in class_identifiers and mod_var_name not in trigger_attributes:
                    trigger_attributes[mod_var_name] = mod_var_src

            if sig_name:
                class_source = _extract_class_source(file_lines, node)

                # Fix #10: raise error on duplicate names
                if sig_name in index:
                    existing = index[sig_name]
                    print(
                        f"[INDEX] ERROR: Duplicate signature name '{sig_name}':\n"
                        f"  1) {Path(existing.file_path).name} "
                        f"(class {existing.class_name})\n"
                        f"  2) {py_file.name} (class {node.name})\n"
                        f"  Keeping first occurrence."
                    )
                    # Keep first, don't overwrite. Log for manual review.
                    continue

                index[sig_name] = SignatureInfo(
                    name=sig_name,
                    class_name=node.name,
                    file_path=str(py_file),
                    source_code=class_source,
                    description=description,
                    severity=severity,
                    categories=categories,
                    ttp=ttp,
                    filter_apinames=filter_apinames,
                    filter_categories=filter_categories,
                    trigger_attributes=trigger_attributes,
                )

    print(
        f"[INDEX] Indexed {len(index)} signature classes from "
        f"{len(py_files)} files ({skipped} skipped due to errors)"
    )
    return index


def count_mark_call_sites(source_code: str) -> Tuple[int, bool]:
    """
    Count distinct mark/mark_call/mark_ioc calls in the AST as a lower bound,
    tracking whether any mark calls are located within loop bodies.
    """
    try:
        tree = ast.parse(source_code)
    except SyntaxError:
        return 1, False

    class MarkVisitor(ast.NodeVisitor):
        def __init__(self):
            self.count = 0
            self.loop_depth = 0
            self.has_loop_marks = False

        def visit_For(self, node):
            self.loop_depth += 1
            self.generic_visit(node)
            self.loop_depth -= 1

        def visit_While(self, node):
            self.loop_depth += 1
            self.generic_visit(node)
            self.loop_depth -= 1

        def visit_Call(self, node):
            is_mark = False
            if isinstance(node.func, ast.Attribute):
                if node.func.attr in ("mark", "mark_call", "mark_ioc"):
                    is_mark = True
            elif isinstance(node.func, ast.Name):
                if node.func.id in ("mark", "mark_call", "mark_ioc"):
                    is_mark = True
            if is_mark:
                self.count += 1
                if self.loop_depth > 0:
                    self.has_loop_marks = True
            self.generic_visit(node)

    visitor = MarkVisitor()
    visitor.visit(tree)
    return max(1, visitor.count), visitor.has_loop_marks

def check_gemini() -> bool:
    """Check if Gemini API is reachable and the configured model is available."""
    if not GEMINI_API_KEY:
        print("[X] ERROR: GEMINI_API_KEY is not configured.")
        return False
    try:
        model_info = client.models.get(model=GEMINI_MODEL)
        if model_info:
            return True
        return False
    except Exception as e:
        print(f"[GEMINI] ERROR: Connectivity check failed: {e}")
        return False


def call_gemini(prompt: str) -> Optional[str]:
    """Call Google Gemini API with structured JSON output enforcement and rate limit backoff."""
    config = types.GenerateContentConfig(
        temperature=GEMINI_TEMPERATURE,
        response_mime_type="application/json",
        system_instruction=(
            "You are a cybersecurity expert evaluating defense mitigations against Cuckoo Sandbox signatures. "
            "Always respond with valid JSON only. Be precise and conservative in your analysis. "
            "When uncertain whether a defense blocks a trigger path, default to SURVIVING."
        ),
    )
    for attempt in range(1, MAX_RETRIES + 2):
        try:
            response = client.models.generate_content(
                model=GEMINI_MODEL,
                contents=prompt,
                config=config,
            )
            if response and response.text:
                return response.text
            else:
                print(f"[GEMINI] WARN: Empty response from model on attempt {attempt}")
        except Exception as e:
            err_str = str(e)
            print(f"[GEMINI] ERROR (Attempt {attempt}/{MAX_RETRIES + 1}): {err_str[:250]}")
            if "429" in err_str or "RESOURCE_EXHAUSTED" in err_str:
                # Extract requested retry delay from Google's error if available
                delay_match = re.search(r"retryDelay':\s*'(\d+(?:\.\d+)?)(?:s)?'", err_str)
                if not delay_match:
                    delay_match = re.search(r"retry after\s*(\d+(?:\.\d+)?)", err_str, re.IGNORECASE)

                if delay_match:
                    wait_time = float(delay_match.group(1)) + 2.0
                else:
                    wait_time = max(35.0, 20.0 * attempt)

                print(f"[GEMINI] Rate limit reached. Waiting {wait_time:.1f}s before retrying...")
                time.sleep(wait_time)
            elif attempt <= MAX_RETRIES:
                time.sleep(2)
    return None




def parse_llm_response(
    response_text: str, initial_severity: int
) -> Optional[dict]:
    """
    Parse and strictly validate the JSON response from the LLM.
    Fix #7: reject invalid formats, do NOT silently repair.
    Fix #8: reject invalid labels, do NOT infer from value.
    """
    if not response_text:
        return None

    text = response_text.strip()

    # Remove markdown code fences if present
    if text.startswith("```"):
        lines = text.split("\n")
        start_idx = 1
        end_idx = len(lines)
        for i in range(len(lines) - 1, 0, -1):
            if lines[i].strip().startswith("```"):
                end_idx = i
                break
        text = "\n".join(lines[start_idx:end_idx])

    # Parse JSON (safe raw_decode to prevent greedy regex bugs)
    try:
        data = json.loads(text)
    except json.JSONDecodeError:
        start = text.find("{")
        if start != -1:
            try:
                decoder = json.JSONDecoder()
                data, _ = decoder.raw_decode(text[start:])
            except json.JSONDecodeError:
                return None
        else:
            return None

    # Ensure reasoning has a default if not present
    if "reasoning" not in data:
        data["reasoning"] = data.get("defense_interpretation", "")

    # Validate required fields exist
    required = [
        "trigger_conditions",
        "defense_interpretation",
        "blocked_paths",
        "surviving_paths",
    ]
    for key in required:
        if key not in data:
            return None

    # Validate types
    if not isinstance(data["trigger_conditions"], list):
        return None
    if not isinstance(data["blocked_paths"], list):
        return None
    if not isinstance(data["surviving_paths"], list):
        return None
    if not isinstance(data["defense_interpretation"], str):
        return None

    # Validate trigger conditions are non-empty
    if len(data["trigger_conditions"]) == 0:
        return None

    # Accept even if blocked + surviving != total.
    # We use the LLM's own blocked/surviving lists for the ratio.
    # If both are empty but triggers exist, treat all as surviving.
    if len(data["blocked_paths"]) == 0 and len(data["surviving_paths"]) == 0:
        data["surviving_paths"] = list(data["trigger_conditions"])

    # Extract and normalize intended_mitigation field
    intended = str(data.get("intended_mitigation", "")).strip().upper()
    if intended not in ("FULL", "PARTIAL", "NONE"):
        if len(data["blocked_paths"]) > 0 and len(data["surviving_paths"]) == 0:
            intended = "FULL"
        elif len(data["blocked_paths"]) == 0:
            intended = "NONE"
        else:
            intended = "PARTIAL"
    data["intended_mitigation"] = intended

    return data


def _extract_path_id(text: str, list_type: str, fallback_idx: int) -> str:
    """Extract normalized path identifier (e.g. 'path 1' -> 'path_1') or clean content key."""
    if not text or not str(text).strip():
        return f"{list_type}_empty_{fallback_idx}"
    m = re.match(r'^(?:path\s*)?(\d+[a-z]?)', str(text).strip(), re.IGNORECASE)
    if m:
        return f"path_{m.group(1).lower()}"
    cleaned = re.sub(r'\s+', ' ', str(text).strip().lower())
    return cleaned[:80] if cleaned else f"{list_type}_fallback_{fallback_idx}"


def compute_result(
    data: dict, initial_severity: int
) -> Tuple[str, int, int, int]:
    """
    Compute mitigation_label and llm_value from parsed LLM response.
    Uses deterministic path ratio formula with strict conservative accounting:
    - FULL requires blocked > 0 AND effective_surviving == 0.
    - Any path in trigger_conditions not explicitly blocked is treated as surviving.
    - Any conflict (path ID appearing in both lists) is treated as surviving.
    - If the model explicitly lists ANY surviving path, it can NEVER be FULL.
    Returns (label, llm_value, total_paths, blocked_count).
    LLM_Value is always in [0, IS]. No negative values.
    """
    total = max(1, len(data.get("trigger_conditions", [])))
    
    blocked_raw = data.get("blocked_paths", [])
    surviving_raw = data.get("surviving_paths", [])
    
    # Extract path IDs and deduplicate surviving paths
    surviving_ids = set()
    unique_surviving = []
    for idx, p in enumerate(surviving_raw):
        pid = _extract_path_id(p, "surviving", idx)
        if pid not in surviving_ids:
            surviving_ids.add(pid)
            unique_surviving.append(p)
    
    # Filter blocked paths: exclude any path ID that also appears in surviving (conflict -> surviving)
    blocked_ids = set()
    clean_blocked = []
    for idx, p in enumerate(blocked_raw):
        pid = _extract_path_id(p, "blocked", idx)
        if pid not in surviving_ids and pid not in blocked_ids:
            blocked_ids.add(pid)
            clean_blocked.append(p)
    
    # Count valid distinct paths bounded by total
    blocked = min(len(clean_blocked), total)
    explicit_surviving = min(len(unique_surviving), total)
    
    # Unaccounted paths: trigger conditions not accounted for in either list
    # Conservatively treated as SURVIVING (worst-case assumption)
    unaccounted = max(0, total - (blocked + explicit_surviving))
    effective_surviving = explicit_surviving + unaccounted
    
    # Compute residual severity using floor+clamp (empirically verified against ground_value.csv)
    if blocked > 0 and effective_surviving == 0:
        # Every trigger path is accounted for and 100% of them are blocked
        return ("FULL", 0, total, blocked)
    elif blocked == 0:
        # Zero paths are blocked -> no mitigation
        return ("NONE", initial_severity, total, 0)
    else:
        # PARTIAL mitigation: residual severity is proportional to surviving fraction
        # Math: RS = round(IS * surviving / total), clamped to [1, IS-1]
        rs = int(round(initial_severity * effective_surviving / total + 1e-9))
        if initial_severity > 1:
            rs = max(1, min(initial_severity - 1, rs))
        else:
            rs = 1
        return ("PARTIAL", rs, total, blocked)


def _format_trigger_attrs(sig_info: SignatureInfo) -> str:
    """Format extracted trigger attributes for the prompt as a compact summary."""
    lines = []
    for attr_name, raw_source in sig_info.trigger_attributes.items():
        if attr_name in ("filter_apinames", "filter_categories"):
            continue  # Already shown separately in header
        preview = raw_source.replace("\n", " ").strip()
        if len(preview) > 120:
            preview = preview[:117] + "..."
        lines.append(f"  {attr_name}: {preview}")
    if not lines:
        return "Other trigger attributes: (see full source code)"
    return "Other trigger attributes (full content in source code below):\n" + "\n".join(lines)

def correct_inconsistent_response(original_data: dict, total_paths: int, row: CsvRow) -> dict:
    """Send targeted correction request if path counts do not match trigger conditions count."""
    print(
        f"  [CORRECT] Triggering targeted path-count correction for {row.signature_name} vs {row.defense} "
        f"({len(original_data.get('blocked_paths', []))} blk + {len(original_data.get('surviving_paths', []))} surv != {total_paths} total)"
    )

    defense_context = _get_defense_context(row.defense, row.defense_description)
    correction_prompt = textwrap.dedent(f"""\
    You are evaluating signature "{row.signature_name}" against defense "{row.defense}".
    You previously identified {total_paths} trigger paths but listed {len(original_data.get('blocked_paths', []))} blocked and {len(original_data.get('surviving_paths', []))} surviving.
    That sums to {len(original_data.get('blocked_paths', [])) + len(original_data.get('surviving_paths', []))}, not {total_paths}.

    Defense Under Evaluation: {row.defense}
    Defense Capabilities & Reference Principles:
    {defense_context}

    Please revise your JSON so that EVERY one of the {total_paths} trigger paths appears in EXACTLY ONE of "blocked_paths" or "surviving_paths" based on whether {row.defense} mitigates it.
    Return ONLY the corrected JSON with the exact same structure.

    Original trigger_conditions: {json.dumps(original_data.get('trigger_conditions', []))}
    Original defense_interpretation: {original_data.get('defense_interpretation', '')}
    Original reasoning: {original_data.get('reasoning', '')}
    """)
    new_response = call_gemini(correction_prompt)
    if new_response:
        corrected = parse_llm_response(new_response, row.initial_severity)
        if corrected and len(corrected["blocked_paths"]) + len(corrected["surviving_paths"]) == len(corrected["trigger_conditions"]):
            print(f"  [CORRECT] Correction succeeded: {len(corrected['blocked_paths'])} blocked, {len(corrected['surviving_paths'])} surviving.")
            return corrected
    # Fallback: treat missing paths as surviving
    print("  [CORRECT] Fallback applied: treating missing paths as surviving.")
    if len(original_data["surviving_paths"]) + len(original_data["blocked_paths"]) < total_paths:
        missing = total_paths - (len(original_data["blocked_paths"]) + len(original_data["surviving_paths"]))
        original_data["surviving_paths"].extend(original_data["trigger_conditions"][-missing:])
    return original_data


def correct_single_path_partial(original_data: dict, row: CsvRow) -> dict:
    """Prompt LLM to decompose a single path into blocked and surviving facets when partial mitigation is intended."""
    print(
        f"  [CORRECT] Reconciling single-path partial contradiction for {row.signature_name} vs {row.defense} (IS={row.initial_severity})"
    )
    defense_context = _get_defense_context(row.defense, row.defense_description)
    reconcile_prompt = textwrap.dedent(f"""\
    You are evaluating signature "{row.signature_name}" against defense "{row.defense}" (Initial Severity IS = {row.initial_severity}).
    You explicitly declared "intended_mitigation": "PARTIAL":
    "{original_data.get('defense_interpretation', '')}"

    However, you provided only 1 trigger path: {json.dumps(original_data.get('trigger_conditions', []))}.
    With only 1 trigger path, the residual severity formula mathematically forces an all-or-nothing score (either 0 FULL or {row.initial_severity} NONE), which contradicts your intended partial mitigation!

    To calculate partial mitigation for an IS >= 2 signature, decompose the threat behavior into at least TWO distinct operational trigger paths:
    1. The blocked/mitigated facet (e.g. anomaly detection alert, containment, blocked payload execution, or neutralized impact) -> put in "blocked_paths".
    2. The surviving/unblocked facet (e.g. underlying Win32 API execution, initial file write, or unmonitored baseline operation) -> put in "surviving_paths".

    Defense Capabilities & Guidance:
    {defense_context}

    Please revise your JSON so that the behavior is decomposed into distinct paths with at least one in "blocked_paths" and at least one in "surviving_paths".
    Return ONLY a valid JSON object:
    {{
      "intended_mitigation": "PARTIAL",
      "trigger_conditions": ["path 1: <blocked/mitigated facet>", "path 2: <surviving facet>"],
      "defense_interpretation": "{original_data.get('defense_interpretation', '')}",
      "blocked_paths": ["path 1: <blocked/mitigated facet>"],
      "surviving_paths": ["path 2: <surviving facet>"]
    }}
    """)
    new_response = call_gemini(reconcile_prompt)
    if new_response:
        corrected = parse_llm_response(new_response, row.initial_severity)
        if corrected and len(corrected["blocked_paths"]) > 0 and len(corrected["surviving_paths"]) > 0:
            print(f"  [CORRECT] Reconciled to {len(corrected['blocked_paths'])} blocked and {len(corrected['surviving_paths'])} surviving paths.")
            return corrected
    return original_data


def evaluate_row(
    row: CsvRow,
    sig_info: Optional[SignatureInfo],
    run_id: str,
) -> EvalResult:
    """Evaluate a single signature-defense pair using the LLM."""

    now = datetime.now(timezone.utc).isoformat()

    # Handle missing signature: synthesize fallback SignatureInfo from row metadata
    if sig_info is None:
        print(
            f"  [+] Synthesizing fallback signature for '{row.signature_name}' "
            f"from data.csv metadata."
        )
        sig_info = SignatureInfo(
            name=row.signature_name,
            class_name=row.signature_name,
            file_path="synthetic",
            source_code=(
                f"# Fallback Signature Definition\n"
                f"# Signature Name: {row.signature_name}\n"
                f"# Description: {row.signature_description}\n"
                f"# Tactics: {row.tactics}\n"
                f"# Technique: {row.technique}\n"
                f"# Initial Severity: {row.initial_severity}\n"
            ),
            description=row.signature_description,
            severity=row.initial_severity,
            categories=[t.strip() for t in row.tactics.split(",") if t.strip()] if row.tactics else [],
            ttp=[row.technique] if row.technique else [],
            filter_apinames=[],
            filter_categories=[],
            trigger_attributes={"description": row.signature_description},
        )

    # Severity reconciliation
    # Bug 9 fix: severity == -1 means non-literal, skip mismatch check for those
    if sig_info.severity == -1:
        severity_mismatch = False  # Cannot compare; don't raise spurious mismatch
    else:
        severity_mismatch = sig_info.severity != row.initial_severity
        if severity_mismatch:
            print(
                f"  [WARN] Severity mismatch: CSV IS={row.initial_severity}, "
                f"Signature severity={sig_info.severity}"
            )

    # Compute mark call sites as lower bound anchor (loop-aware)
    mark_count, has_loop_marks = count_mark_call_sites(sig_info.source_code)
    if has_loop_marks:
        mark_count_note = (
            f"There are at least {mark_count} distinct `mark_*` call site(s) in this code "
            f"(including call sites inside loops iterating over indicator lists / regex patterns). "
            f"Treat this as a baseline lower bound; identify each distinct indicator / pattern category as an independent trigger path."
        )
    else:
        mark_count_note = (
            f"There are at least {mark_count} distinct `mark_*` call site(s) in this code. "
            f"Use this count as a baseline anchor for independent trigger paths."
        )

    # Build structured prompt
    api_list = (
        ", ".join(sig_info.filter_apinames)
        if sig_info.filter_apinames
        else "None extracted (see source code)"
    )
    filter_categories_list = (
        ", ".join(sig_info.filter_categories)
        if sig_info.filter_categories
        else "None specified (all categories)"
    )
    trigger_attrs_text = _format_trigger_attrs(sig_info)

    # Bug 9 fix: show N/A for non-literal severity instead of misleading 0
    if sig_info.severity == -1:
        code_severity_str = "N/A (non-literal constant, see source code)"
    else:
        code_severity_str = str(sig_info.severity)

    sig_desc = row.signature_description or sig_info.description or "N/A"
    tactics = row.tactics or (", ".join(sig_info.ttp) if sig_info.ttp else "None listed")
    technique = row.technique or "None listed"
    def_desc = row.defense_description or "N/A"

    defense_context = _get_defense_context(row.defense, row.defense_description)
    prompt = PROMPT_TEMPLATE.format(
        signature_name=sig_info.name,
        signature_description=sig_desc,
        tactics=tactics,
        technique=technique,
        categories=(
            ", ".join(sig_info.categories) if sig_info.categories else "N/A"
        ),
        code_severity=code_severity_str,
        mark_count_note=mark_count_note,
        api_list=api_list,
        filter_categories_list=filter_categories_list,
        trigger_attrs_text=trigger_attrs_text,
        source_code=sig_info.source_code,
        defense_name=row.defense,
        defense_description=def_desc,
        defense_guidance=defense_context,
    )

    # Call LLM with retries (Fix #8: reject invalid, retry)
    last_response = None
    for attempt in range(1, MAX_RETRIES + 2):
        response_text = call_gemini(prompt)
        last_response = response_text

        if response_text:
            parsed = parse_llm_response(response_text, row.initial_severity)
            if parsed:
                # Consistency correction if path lists sum does not match trigger_conditions
                total_t = len(parsed["trigger_conditions"])
                if len(parsed["blocked_paths"]) + len(parsed["surviving_paths"]) != total_t:
                    parsed = correct_inconsistent_response(parsed, total_t, row)

                # Reconcile single-path partial contradiction for IS >= 2 signatures
                # ONLY if the model ITSELF explicitly declared "intended_mitigation": "PARTIAL".
                # If intended_mitigation is FULL or NONE, a single path is completely valid and preserved!
                if row.initial_severity >= 2 and len(parsed["trigger_conditions"]) == 1:
                    intended = parsed.get("intended_mitigation", "").upper()
                    if intended == "PARTIAL":
                        parsed = correct_single_path_partial(parsed, row)

                label, llm_value, total_paths, blocked_count = compute_result(
                    parsed, row.initial_severity
                )
                return EvalResult(
                    signature_name=row.signature_name,
                    defense=row.defense,
                    initial_severity=row.initial_severity,
                    signature_severity=sig_info.severity,
                    severity_mismatch=severity_mismatch,
                    llm_value=llm_value,
                    mitigation_label=label,
                    total_trigger_paths=total_paths,
                    blocked_trigger_paths=blocked_count,
                    trigger_conditions=json.dumps(
                        parsed["trigger_conditions"]
                    ),
                    defense_interpretation=parsed["defense_interpretation"],
                    blocked_paths=json.dumps(parsed["blocked_paths"]),
                    surviving_paths=json.dumps(parsed["surviving_paths"]),
                    reasoning=parsed["reasoning"],
                    run_id=run_id,
                    timestamp=now,
                    model=GEMINI_MODEL,
                )

        if attempt <= MAX_RETRIES:
            print(
                f"  [!] Attempt {attempt} failed (invalid response), "
                f"retrying ({MAX_RETRIES - attempt + 1} left)..."
            )
            time.sleep(1)

    # All retries exhausted
    print(f"  [X] All {MAX_RETRIES + 1} attempts failed -> UNKNOWN")
    return EvalResult(
        signature_name=row.signature_name,
        defense=row.defense,
        initial_severity=row.initial_severity,
        signature_severity=sig_info.severity if sig_info else 0,
        severity_mismatch=severity_mismatch,
        llm_value=row.initial_severity,
        mitigation_label="UNKNOWN",
        total_trigger_paths=0,
        blocked_trigger_paths=0,
        trigger_conditions="",
        defense_interpretation="",
        blocked_paths="",
        surviving_paths="",
        reasoning=(
            f"LLM failed to produce valid JSON after {MAX_RETRIES + 1} "
            f"attempts. Last raw response: "
            f"{(last_response or 'None')[:300]}"
        ),
        run_id=run_id,
        timestamp=now,
        model=GEMINI_MODEL,
    )

def init_results_csv(results_path: Path) -> None:
    """Write the header to results.csv if it doesn't exist or is empty."""
    if not results_path.exists() or results_path.stat().st_size == 0:
        with open(results_path, "w", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            writer.writerow(RESULTS_COLUMNS)
        print(f"[RESULTS] Created {results_path.name} with header")


def append_result(results_path: Path, result: EvalResult) -> None:
    """Append a single result row to results.csv."""
    with open(results_path, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(
            [
                result.signature_name,
                result.defense,
                result.initial_severity,
                result.signature_severity,
                result.severity_mismatch,
                result.llm_value,
                result.mitigation_label,
                result.total_trigger_paths,
                result.blocked_trigger_paths,
                result.trigger_conditions,
                result.defense_interpretation,
                result.blocked_paths,
                result.surviving_paths,
                result.reasoning,
                result.run_id,
                result.timestamp,
                result.model,
            ]
        )


def load_completed_keys(results_path: Path) -> Set[Tuple[str, str]]:
    """
    Load set of (Signature_Name, Defense) keys already processed.
    Fix #6: key-based resume instead of row-count-based.
    """
    completed: Set[Tuple[str, str]] = set()
    if not results_path.exists():
        return completed
    try:
        with open(results_path, "r", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            for r in reader:
                key = (r["Signature_Name"], r["Defenses"])
                completed.add(key)
    except (KeyError, csv.Error) as e:
        print(f"[RESUME] WARN: Could not read existing results: {e}")
        print("[RESUME] Starting fresh.")
        return set()
    return completed

def main():
    print("=" * 70)
    print("  Cuckoo Signature--Defense Mitigation Evaluator v2 (Gemini)")
    print(f"  Model: {GEMINI_MODEL} | Temperature: {GEMINI_TEMPERATURE}")
    print("=" * 70)
    print()

    # Generate run ID (Fix #12)
    run_id = str(uuid.uuid4())[:8]
    print(f"[RUN] Run ID: {run_id}")
    print()

    # Check Gemini API connectivity
    print("[1/4] Checking Gemini API connectivity...")
    if not check_gemini():
        print(
            "\n[X] ERROR: Cannot connect to Gemini API."
            "\n  Please verify your GEMINI_API_KEY in .env or environment variables."
        )
        sys.exit(1)
    print(f"[GEMINI] [OK] Connected. Model '{GEMINI_MODEL}' verified.\n")

    # Parse CSV (Fix #5: header-based)
    print("[2/4] Parsing data.csv...")
    rows = parse_csv(DATA_CSV)
    if not rows:
        print("[X] ERROR: No valid rows found in data.csv")
        sys.exit(1)
    print()

    # Build index (Fix #9: recursive, Fix #10: duplicate detection)
    print("[3/4] Building signature index from windows/**/*.py...")
    sig_index = build_signature_index(WINDOWS_DIR)
    print()

    # Check which signatures from CSV exist in the index
    unique_sigs = set(r.signature_name for r in rows)
    found = unique_sigs & set(sig_index.keys())
    missing = unique_sigs - set(sig_index.keys())
    print(
        f"[CHECK] {len(found)}/{len(unique_sigs)} unique signatures "
        f"found in index"
    )
    if missing:
        print(f"[CHECK] [!] Missing signatures: {sorted(missing)}")
    print()

    # Key-based resume (Fix #6)
    init_results_csv(RESULTS_CSV)
    completed_keys = load_completed_keys(RESULTS_CSV)
    if completed_keys:
        print(
            f"[RESUME] Found {len(completed_keys)} already-processed pairs "
            f"in {RESULTS_CSV.name}"
        )
    print()

    # Evaluate
    total = len(rows)
    to_process = sum(
        1
        for r in rows
        if (r.signature_name, r.defense) not in completed_keys
    )
    print(
        f"[4/4] Evaluating {to_process} remaining rows "
        f"(of {total} total)..."
    )
    print("-" * 70)

    start_time = time.time()
    processed = 0
    for i, row in enumerate(rows):
        row_num = i + 1
        key = (row.signature_name, row.defense)

        if key in completed_keys:
            continue  # Skip already-processed (Fix #6)

        sig_info = sig_index.get(row.signature_name)
        sig_file = (
            Path(sig_info.file_path).name if sig_info else "NOT FOUND"
        )

        print(
            f"[{row_num}/{total}] "
            f"Signature: {row.signature_name} | "
            f"Defense: {row.defense} | "
            f"IS={row.initial_severity} | "
            f"File: {sig_file}"
        )

        result = evaluate_row(row, sig_info, run_id)
        append_result(RESULTS_CSV, result)
        processed += 1

        print(
            f"  -> LLM_Value={result.llm_value}, "
            f"Label={result.mitigation_label}, "
            f"Paths={result.total_trigger_paths} total / "
            f"{result.blocked_trigger_paths} blocked"
        )
        time.sleep(ROW_DELAY)

    elapsed = time.time() - start_time
    print("-" * 70)
    print(f"\n[OK] Done! Processed {processed} rows in {elapsed:.1f}s")
    print(f"  Results saved to: {RESULTS_CSV}")
    print(f"  Run ID: {run_id}")

    # Summary
    full_count = 0
    partial_count = 0
    none_count = 0
    unknown_count = 0
    mismatches = 0
    with open(RESULTS_CSV, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for r in reader:
            label = r.get("Mitigation_Label", "").upper()
            if label == "FULL":
                full_count += 1
            elif label == "PARTIAL":
                partial_count += 1
            elif label == "NONE":
                none_count += 1
            else:
                unknown_count += 1
            if r.get("Severity_Mismatch", "").lower() == "true":
                mismatches += 1

    print(f"\n  Summary:")
    print(f"    FULL:    {full_count}")
    print(f"    PARTIAL: {partial_count}")
    print(f"    NONE:    {none_count}")
    print(f"    UNKNOWN: {unknown_count}")
    if mismatches:
        print(f"    Severity mismatches: {mismatches}")


if __name__ == "__main__":
    main()
